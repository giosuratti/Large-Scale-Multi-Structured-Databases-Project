LOAD CSV WITH HEADERS FROM 'file:///nodes_specialization.csv' AS row
MERGE (sp:Specialization {name: row.name});

LOAD CSV WITH HEADERS FROM 'file:///nodes_symptom.csv' AS row
MERGE (s:Symptom {name: row.name});

LOAD CSV WITH HEADERS FROM 'file:///nodes_disease.csv' AS row
MERGE (d:Disease {name: row.name});

LOAD CSV WITH HEADERS FROM 'file:///rel_has_symptom.csv' AS row
MATCH (d:Disease {name: row.start})
MATCH (s:Symptom {name: row.end})
MERGE (s)-[:INDICATES_DISEASE {weight: row.weight}]->(d);

LOAD CSV WITH HEADERS FROM 'file:///rel_treated_by.csv' AS row
MATCH (d:Disease {name: row.start})
MATCH (sp:Specialization {name: row.end})
MERGE (d)-[:TREATED_BY]->(sp);

CREATE INDEX symptom_name_index IF NOT EXISTS
FOR (sym:Symptom) ON (sym.name);

CREATE INDEX disease_name_index IF NOT EXISTS
FOR (d:Disease) ON (d.name);

CREATE INDEX doctor_npi_index IF NOT EXISTS
FOR (d:Doctor) ON (d.NPI);

CREATE INDEX specialization_name_index IF NOT EXISTS
FOR (s:Specialization) ON (s.name);

LOAD CSV WITH HEADERS FROM "file:///dottori_output_neo4j.csv" AS row
WITH row.Specialization AS spec
WHERE spec <> ""
MERGE (:Specialization {name: spec});

LOAD CSV WITH HEADERS FROM "file:///dottori_output_neo4j.csv" AS row
MERGE (d:Doctor {NPI: row.NPI})
SET d.firstName = row.`Provider First Name`,
    d.lastName = row.`Provider Last Name`,
    d.city = row.City,
    d.phone = row.Telephone;

LOAD CSV WITH HEADERS FROM "file:///dottori_output_neo4j.csv" AS row
MATCH (d:Doctor {NPI: row.NPI})
WITH d, row.Specialization AS spec
WHERE spec <> ""
MATCH (s:Specialization {name: spec})
CREATE (s)-[:HAS_DOCTOR]->(d);
